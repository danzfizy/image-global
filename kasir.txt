// Data Model
class CategoryModel {
  String id;
  String name;
  CategoryModel({required this.id, required this.name});
}

class MenuItemModel {
  String id;
  String name;
  double price;
  String categoryId;

  MenuItemModel({
    required this.id,
    required this.name,
    required this.price,
    required this.categoryId,
  });

  // Fungsi membuat singkatan 2 huruf dari nama menu
  String get shortName {
    List<String> words = name.trim().split(RegExp(r'\s+'));
    if (words.length >= 2) {
      return (words[0][0] + words[1][0]).toUpperCase();
    } else if (name.length >= 2) {
      return name.substring(0, 2).toUpperCase();
    } else if (name.isNotEmpty) {
      return name.toUpperCase();
    }
    return '--';
  }
}

class OrderItemModel {
  MenuItemModel item;
  int quantity;

  OrderItemModel({required this.item, required this.quantity});
  double get totalPrice => item.price * quantity;
}

  double get grandTotal {
    return cart.fold(0, (sum, item) => sum + item.totalPrice);
  }

  // Logika Cart
  void _addToCart(MenuItemModel item) {
    setState(() {
      int index = cart.indexWhere((element) => element.item.id == item.id);
      if (index != -1) {
        cart[index].quantity++;
      } else {
        cart.add(OrderItemModel(item: item, quantity: 1));
      }
      _calculateChange();
    });
  }

  void _decreaseItemQuantity(MenuItemModel item) {
    setState(() {
      int index = cart.indexWhere((element) => element.item.id == item.id);
      if (index != -1) {
        if (cart[index].quantity > 1) {
          cart[index].quantity--;
        } else {
          cart.removeAt(index);
        }
      }
      _calculateChange();
    });
  }

      setState(() {
        changeAmount = cash - grandTotal;

    } else {
      setState(() {

  // Fitur Kirim Struk ke Telegram Channel / Bot
  Future<void> _sendReceiptToTelegram({required bool isKitchen}) async {
    if (telegramBotToken == 'YOUR_TELEGRAM_BOT_TOKEN') {
      ScaffoldMessenger.of(context).showSnackBar(
        const SnackBar(content: Text('Harap isi token Bot Telegram & Chat ID pada kode lebih dulu!')),
      );
    }

    String title = isKitchen ? "🧾 **STRUK DAPUR - KASIR CAFE**" : "🧾 **STRUK PEMBAYARAN - KASIR CAFE**";
    String text = "$title\n";
    text += "-----------------------------------\n";
    for (var item in cart) {
      if (isKitchen) {
        text += "• ${item.item.name} x${item.quantity}\n";
      } else {
        text += "• ${item.item.name} (${item.quantity}x) = Rp ${item.totalPrice.toStringAsFixed(0)}\n";
      }
    }
    text += "-----------------------------------\n";

    if (!isKitchen) {
      text += "TOTAL : Rp ${grandTotal.toStringAsFixed(0)}\n";
      text += "METODE: $paymentMethod\n";
      if (paymentMethod == 'Tunai') {
        double cash = double.tryParse(cashInputController.text) ?? 0;
        text += "BAYAR : Rp ${cash.toStringAsFixed(0)}\n";
        text += "KEMBALI: Rp ${changeAmount.toStringAsFixed(0)}\n";
      }
    }

    final url = Uri.parse('https://api.telegram.org/bot$telegramBotToken/sendMessage');
    try {
      final response = await http.post(
        url,
        headers: {'Content-Type': 'application/json'},
        body: jsonEncode({
          'chat_id': telegramChatId,
          'text': text,
          'parse_mode': 'Markdown',
        }),
      );

      if (response.statusCode == 200) {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text(isKitchen ? 'Struk dapur terkirim ke Telegram!' : 'Struk pembayaran terkirim ke Telegram!')),
        );
      } else {
        ScaffoldMessenger.of(context).showSnackBar(
          SnackBar(content: Text('Gagal mengirim ke Telegram: ${response.body}')),
        );
      }
    } catch (e) {
      ScaffoldMessenger.of(context).showSnackBar(
        SnackBar(content: Text('Terjadi error: $e')),
      );
    }
  }

// Halaman 4. Add Category
class AddCategoryScreen extends StatefulWidget {
  final List<CategoryModel> categories;
  final Function(List<CategoryModel>) onUpdate;

  const AddCategoryScreen({super.key, required this.categories, required this.onUpdate});

  @override
  State<AddCategoryScreen> createState() => _AddCategoryScreenState();
}

class _AddCategoryScreenState extends State<AddCategoryScreen> {
  late List<CategoryModel> categoriesList;
  final TextEditingController categoryController = TextEditingController();

  @override
  void initState() {
    super.initState();
    categoriesList = List.from(widget.categories);
  }

  void _addCategory() {
    if (categoryController.text.trim().isEmpty) return;
    setState(() {
      categoriesList.add(CategoryModel(
        id: 'cat_${DateTime.now().millisecondsSinceEpoch}',
        name: categoryController.text.trim(),
      ));
      categoryController.clear();
    });
    widget.onUpdate(categoriesList);
  }

  void _deleteCategory(int index) {
    if (categoriesList[index].id == 'cat_all') return; // Jangan hapus kategori "Semua"
    setState(() {
      categoriesList.removeAt(index);
    });
    widget.onUpdate(categoriesList);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Kelola Kategori'), backgroundColor: Colors.brown, foregroundColor: Colors.white),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            Row(
              children: [
                Expanded(
                  child: TextField(
                    controller: categoryController,
                    decoration: const InputDecoration(
                      labelText: 'Nama Kategori Baru',
                      border: OutlineInputBorder(),
                    ),
                  ),
                ),
                const SizedBox(width: 10),
                ElevatedButton(
                  style: ElevatedButton.styleFrom(backgroundColor: Colors.brown, foregroundColor: Colors.white),
                  onPressed: _addCategory,
                  child: const Text('Tambah'),
                )
              ],
            ),
            const SizedBox(height: 20),
            Expanded(
              child: ListView.builder(
                itemCount: categoriesList.length,
                itemBuilder: (context, index) {
                  final cat = categoriesList[index];
                  return Card(
                    child: ListTile(
                      title: Text(cat.name),
                      trailing: cat.id == 'cat_all'
                          ? null
                          : IconButton(
                              icon: const Icon(Icons.delete, color: Colors.red),
                              onPressed: () => _deleteCategory(index),
                            ),
                    ),
                  );
                },
              ),
            ),
          ],
        ),
      ),
    );
  }
}

// Halaman 5. Add / Manage Menu
class AddMenuScreen extends StatefulWidget {
  final List<MenuItemModel> menuItems;
  final List<CategoryModel> categories;
  final Function(List<MenuItemModel>) onUpdate;

  const AddMenuScreen({
    super.key,
    required this.menuItems,
    required this.categories,
    required this.onUpdate,
  });

  @override
  State<AddMenuScreen> createState() => _AddMenuScreenState();
}

class _AddMenuScreenState extends State<AddMenuScreen> {
  late List<MenuItemModel> menuList;
  String searchQuery = '';

  @override
  void initState() {
    super.initState();
    menuList = List.from(widget.menuItems);
  }

  List<MenuItemModel> get filteredList {
    if (searchQuery.isEmpty) return menuList;
    return menuList.where((item) => item.name.toLowerCase().contains(searchQuery.toLowerCase())).toList();
  }

  void _showFormDialog({MenuItemModel? itemToEdit}) {
    final nameController = TextEditingController(text: itemToEdit?.name ?? '');
    final priceController = TextEditingController(text: itemToEdit != null ? itemToEdit.price.toStringAsFixed(0) : '');
    String selectedCatId = itemToEdit?.categoryId ?? (widget.categories.length > 1 ? widget.categories[1].id : widget.categories[0].id);

    showDialog(
      context: context,
      builder: (context) {
        return AlertDialog(
          title: Text(itemToEdit == null ? 'Tambah Menu Baru' : 'Edit Menu'),
          content: SingleChildScrollView(
            child: Column(
              mainAxisSize: MainAxisSize.min,
              children: [
                TextField(
                  controller: nameController,
                  decoration: const InputDecoration(labelText: 'Nama Menu'),
                ),
                TextField(
                  controller: priceController,
                  keyboardType: TextInputType.number,
                  decoration: const InputDecoration(labelText: 'Harga (Rp)'),
                ),
                const SizedBox(height: 12),
                DropdownButtonFormField<String>(
                  value: selectedCatId,
                  decoration: const InputDecoration(labelText: 'Kategori'),
                  items: widget.categories.where((c) => c.id != 'cat_all').map((cat) {
                    return DropdownMenuItem(value: cat.id, child: Text(cat.name));
                  }).toList(),
                  onChanged: (val) {
                    if (val != null) selectedCatId = val;
                  },
                )
              ],
            ),
          ),
          actions: [
            TextButton(onPressed: () => Navigator.pop(context), child: const Text('Batal')),
            ElevatedButton(
              onPressed: () {
                if (nameController.text.isEmpty || priceController.text.isEmpty) return;
                setState(() {
                  if (itemToEdit == null) {
                    menuList.add(MenuItemModel(
                      id: 'm_${DateTime.now().millisecondsSinceEpoch}',
                      name: nameController.text.trim(),
                      price: double.tryParse(priceController.text) ?? 0,
                      categoryId: selectedCatId,
                    ));
                  } else {
                    itemToEdit.name = nameController.text.trim();
                    itemToEdit.price = double.tryParse(priceController.text) ?? 0;
                    itemToEdit.categoryId = selectedCatId;
                  }
                });
                widget.onUpdate(menuList);
                Navigator.pop(context);
              },
              child: const Text('Simpan'),
            )
          ],
        );
      },
    );
  }

  void _deleteMenu(MenuItemModel item) {
    setState(() {
      menuList.removeWhere((element) => element.id == item.id);
    });
    widget.onUpdate(menuList);
  }

  @override
  Widget build(BuildContext context) {
    return Scaffold(
      appBar: AppBar(title: const Text('Kelola Menu'), backgroundColor: Colors.brown, foregroundColor: Colors.white),
      floatingActionButton: FloatingActionButton(
        backgroundColor: Colors.brown,
        onPressed: () => _showFormDialog(),
        child: const Icon(Icons.add, color: Colors.white),
      ),
      body: Padding(
        padding: const EdgeInsets.all(16.0),
        child: Column(
          children: [
            // Search Bar
            TextField(
              decoration: const InputDecoration(
                labelText: 'Cari Menu...',
                prefixIcon: Icon(Icons.search),
                border: OutlineInputBorder(),
              ),
              onChanged: (val) {
                setState(() {
                  searchQuery = val;
                });
              },
            ),
            const SizedBox(height: 16),
            Expanded(
              child: ListView.builder(
                itemCount: filteredList.length,
                itemBuilder: (context, index) {
                  final item = filteredList[index];
                  final catName = widget.categories.firstWhere((c) => c.id == item.categoryId, orElse: () => CategoryModel(id: '', name: '-')).name;
                  return Card(
                    child: ListTile(
                      leading: CircleAvatar(
                        backgroundColor: Colors.brown,
                        child: Text(item.shortName, style: const TextStyle(color: Colors.white)),
                      ),
                      title: Text(item.name),
                      subtitle: Text('Rp ${item.price.toStringAsFixed(0)} | Kategori: $catName'),
                      trailing: Row(
                        mainAxisSize: MainAxisSize.min,
                        children: [
                          IconButton(icon: const Icon(Icons.edit, color: Colors.blue), onPressed: () => _showFormDialog(itemToEdit: item)),
                          IconButton(icon: const Icon(Icons.delete, color: Colors.red), onPressed: () => _deleteMenu(item)),
                        ],
                      ),
                    ),
                  );
                },
              ),
            ),
          ],
        ),
      ),
    );
  }
}
